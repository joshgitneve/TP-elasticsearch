4.4. Quelles sont les 10 offres les plus consultées (labels.offre_id, requêtes GET avec code 200) ? 

FROM logs-web-default
| WHERE @timestamp >= ?_tstart AND @timestamp < ?_tend
| WHERE http.request.method == "GET" AND http.response.status_code == 200
| WHERE labels.offre_id IS NOT NULL
| STATS vues = COUNT(*) BY labels.offre_id
| SORT vues DESC
| LIMIT 10
