URLs touchés:
FROM logs-web-default
| WHERE @timestamp >= "2026-09-28T12:00:00Z" AND @timestamp < "2026-09-28T12:45:00Z"
| WHERE http.response.status_code >= 500
| STATS erreurs = COUNT(*) BY url.original
| SORT erreurs DESC

pas touchés:
FROM logs-web-default
| WHERE @timestamp >= "2026-09-28T12:00:00Z" AND @timestamp < "2026-09-28T12:45:00Z"
| EVAL est_erreur = CASE(http.response.status_code >= 500, 1, 0)
| STATS total = COUNT(*), erreurs = SUM(est_erreur) BY url.path
| SORT erreurs DESC

3. le nombre de réponses en erreur et la durée de l'incident ;
FROM logs-web-default
| WHERE @timestamp >= "2026-09-28T12:00:00Z" AND @timestamp < "2026-09-28T12:45:00Z"
| WHERE http.response.status_code >= 500
| STATS nb_erreurs = COUNT(*), debut = MIN(@timestamp), fin = MAX(@timestamp)
| EVAL duree_minutes = DATE_DIFF("minute", debut, fin)
